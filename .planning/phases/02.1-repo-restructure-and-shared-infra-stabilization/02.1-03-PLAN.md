    ```python
    """Tests for src.infra.multiprocess.split_work."""
    import pytest

    from src.infra.multiprocess import split_work

    def test_split_even():
        """6 items across 3 workers -> 2 each."""
        chunks = split_work([1, 2, 3, 4, 5, 6], 3)
        assert len(chunks) == 3
        assert all(len(c) == 2 for c in chunks)
        # All items present
        assert sorted(sum(chunks, [])) == [1, 2, 3, 4, 5, 6]

    def test_split_uneven():
        """7 items across 3 workers -> sizes 3, 2, 2 (round-robin)."""
        chunks = split_work(list(range(7)), 3)
        sizes = sorted([len(c) for c in chunks])
        assert sizes == [2, 2, 3]

    def test_split_more_workers_than_items():
        """2 items across 5 workers -> 2 chunks with 1 item, 3 empty."""
        chunks = split_work(["a", "b"], 5)
        assert len(chunks) == 5
        non_empty = [c for c in chunks if c]
        assert len(non_empty) == 2

    def test_split_single_worker():
        chunks = split_work([1, 2, 3], 1)
        assert chunks == [[1, 2, 3]]

    def test_split_empty_items():
        chunks = split_work([], 3)
        assert chunks == [[], [], []]

    def test_split_zero_workers_raises():
        with pytest.raises(ValueError):
            split_work([1, 2], 0)

    def test_split_preserves_all_items():
        """No items lost or duplicated."""
        items = list(range(100))
        chunks = split_work(items, 7)
        flat = sum(chunks, [])
        assert sorted(flat) == items
    ```
